\begin{frame}
\frametitle{The main idea}
Consider the Helmholtz equation in $\mathbb{R}^2$
\begin{equation*}
	\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
and the change of coordinates \(u=3x,v=3y\).
\newline
\newline
\pause
What form does the Helmholtz equation take in the new coordinates \(u,v\)?
\pause
\begin{equation*}
\partial_x=\frac{du}{dx}\cdot\partial_u=3\partial_u\quad
\partial_y=\frac{dv}{dy}\cdot\partial_v=3\partial_v
\end{equation*}
\pause
Then the equation becomes:
\begin{equation*}
	\left(3^2\partial^2_u+3^2\partial^2_v+k^2\right)f=0
\end{equation*}
\begin{equation*}
	\highlightcyan{\left(\partial^2_u+\partial^2_v+(k/3)^2\right)f=0}
\end{equation*}
\begin{equation*}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{Less trivial example}
Consider the Helmholtz equation in $\mathbb{R}^2$
\begin{equation*}
	\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
and the change of coordinates \(u=\frac{x}{x^2+y^2}, v=\frac{y}{x^2+y^2}\).
\newline
\newline
\pause
What form does the Helmholtz equation take in the new coordinates \(u,v\)?
\pause
\begin{equation*}
	\partial_x=\dd{u}{x}\partial_u + \dd{v}{x}\partial_v \qquad
	\partial_y=\dd{u}{y}\partial_u + \dd{v}{y}\partial_v,
\end{equation*}
\pause
After some algebra the equation becomes:
\begin{equation*}
\left(
\left(\left(\dd{u}{x}\right)^2
+\left(\dd{u}{y}\right)^2\right)
\left(\partial_u^2+\partial_v^2\right)+ k^2\right)f=0
\end{equation*}
\begin{equation*}
\highlightcyan{\left(
	\partial_u^2+\partial_v^2
	+\left(
	k\middle/\sqrt{\left(\dd{u}{x}\right)^2
	+\left(\dd{u}{y}\right)^2}
	\right)^2
\right)f=0}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{Less trivial example}
Substituting
\begin{equation*}\label{partialu}
	\dd{u}{x}=\frac{y^2-x^2}{(x^2+y^2)^2} \qquad
	\dd{u}{y}=-\frac{2xy}{(x^2+y^2)^2},
\end{equation*}
in the equation
\begin{equation*}
\left(
	\partial_u^2+\partial_v^2
	+\left(
	k\middle/\sqrt{\left(\dd{u}{x}\right)^2
	+\left(\dd{u}{y}\right)^2}
	\right)^2
\right)f=0,
\end{equation*}
and doing some more algebra we get:
\begin{equation*}
\highlightcyan{\left(
	\partial_u^2
	+\partial_v^2
	+\left(k\left(x^2+y^2\right)\right)^2
\right)f=0}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{A non-example}
Not all transformations can be achieved by modifying \(k\).

Consider for example \( u=2x,v=3y\).
\pause

In this case:
\begin{equation*}
	\partial_x=2\partial_u \qquad
	\partial_y=3\partial_v,
\end{equation*}
\pause
and replacing in the original Helmholtz equation:
\begin{equation*}
\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
\pause

we get:
\begin{equation*}
	\highlightyellow{(4\partial_u^2 + 9\partial_v^2 + k^2)f=0,}
\end{equation*}
\pause
so we can not rewrite this equation to have the same form as the original
Helmholtz equation.
\end{frame}

\begin{frame}
\frametitle{Maxwell's equations}
\begin{align*}
\nabla \cdot \boldsymbol{E} &= \frac{\rho}{\varepsilon_0}\\
\nabla \cdot \boldsymbol{B} &= 0\\
\nabla \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t}\\
\nabla \times \boldsymbol{B} &= \frac{1}{c^2}\cdot\frac{\partial \boldsymbol{E}}{\partial t} + \mu_0 \boldsymbol{J}
\end{align*}
\pause
In vacuum we have \(\varepsilon_0\boldsymbol{E}=\boldsymbol{D}\), \(\boldsymbol{B}=\mu_0\boldsymbol{H}\),
\(\mu_0\varepsilon_0=\frac{1}{c^2}\), then
\begin{align*}
\nabla \cdot \boldsymbol{D} &= \rho\\
\nabla \cdot \boldsymbol{B} &= 0\\
\nabla \times \boldsymbol{E} &= -\frac{\partial \boldsymbol{B}}{\partial t}\\
\nabla \times \boldsymbol{H} &=  \frac{\partial \boldsymbol{D}}{\partial t} + \boldsymbol{J}
\end{align*}\label{Max}
In dielectric materials we have \(\varepsilon^{ij}\boldsymbol{E}=\boldsymbol{D}\) and \(\boldsymbol{B}=\mu^{ij}\boldsymbol{H}\).
(Slide \ref{MMax})
\end{frame}

\begin{frame}
\frametitle{Maxwell's equations in arbitrary coordinates (empty space)}
\begin{align*}
\nabla \cdot \boldsymbol{E} &= \frac{\rho}{\varepsilon_0} &\frac{1}{\sqrt{g}}\partial_i\left(\sqrt{g}E^i\right) &= \frac{\rho}{\varepsilon_0}\\
\nabla \cdot \boldsymbol{H} &= 0  		             &\frac{1}{\sqrt{g}}\partial_i\left(\sqrt{g}H^i\right) &=0 \\
\nabla\times \boldsymbol{E} &=-\mu_0\frac{\partial \boldsymbol{H}}{\partial t} &\epsilon^{ijk} \partial_jE_k &= -\mu_0\frac{\partial H^i}{\partial t}\\
\nabla\times \boldsymbol{H} &=\varepsilon_0\frac{\partial \boldsymbol{E}}{\partial t} + \boldsymbol{J} &\epsilon^{ijk} \partial_jH_k &= \varepsilon_0\frac{\partial E^i}{\partial t} + J^i
\end{align*}

\end{frame}

\begin{frame}
\frametitle{Back to  right-handed Cartesian}\label{MMax}
\begin{align*}
\partial_i\left(\sqrt{g}g^{ij}E_j\right) &= \frac{\sqrt{g}\rho}{\varepsilon_0}
&\partial_i D^i&=\varrho\\
\partial_i\left(\sqrt{g}g^{ij}H_j\right) &=0
&\partial_i B^i&=0\\
[ijk]\partial_jE_k &= -\mu_0\frac{\partial\left(\pm \sqrt{g}g^{ij}H_j\right)}{\partial t}
&[ijk]\partial_jE_k&=-\frac{\partial B^i}{\partial t}\\
[ijk]\partial_jH_k &= \varepsilon_0\frac{\partial \left(\pm \sqrt{g}g^{ij}E_j\right)}{\partial t} \pm \sqrt{g}J^i
 &[ijk]\partial_jH_k&=\frac{\partial D^i}{\partial t}+\mathcal{J}^i
\end{align*}
Where \(\varrho=\sqrt{g}\rho\), \(\mathcal{J}^i=\pm\sqrt{g}J^i\) and
we have the constitutive equations:
\[D^i=\varepsilon_0\varepsilon^{ij}E_j,\quad B^i=\mu_0\mu^{ij}H_j\]
\[\varepsilon^{ij}=\mu^{ij}=\pm\sqrt{g}g^{ij}.\]
Remember the equations in slide \ref{Max}
\end{frame}

\begin{frame}
\frametitle{Formula for general coordinates}
\begin{align*}
\frac{1}{\sqrt{\gamma}}\partial_i\left(\sqrt{\gamma}E^i\right) &= \frac{\rho}{\varepsilon_0}
&\partial_i\left(\sqrt{\gamma}D^i\right)&=\sqrt{\gamma}\varrho\\
\frac{1}{\sqrt{\gamma}}\partial_i\left(\sqrt{\gamma}H^i\right) &=0
&\partial_i\left(\sqrt{\gamma}B^i\right)&=0\\
[ijk] \partial_jE_k &= -\mu_0\frac{\partial \left(\sqrt{\gamma} H^i\right)}{\partial t}
&[ijk] \partial_jE_k &= -\frac{\partial \left(\sqrt{\gamma} B^i\right)}{\partial t}\\
[ijk] \partial_jH_k &= \varepsilon_0\frac{\partial \left(\sqrt{\gamma} E^i\right)}{\partial t} + \sqrt{\gamma}J^i
&[ijk] \partial_jH_k &= \frac{\partial \left(\sqrt{\gamma} D^i\right)}{\partial t}\\
&&&+ \sqrt{\gamma}J^i
\end{align*}
In this case we have the constitutive equations:
\[D^i=\varepsilon_0\varepsilon^{ij}E_j,\quad B^i=\mu_0\mu^{ij}H_j\]
\[\varepsilon^{ij}=\mu^{ij}=\pm\frac{\sqrt{g}}{\sqrt{\gamma}}g^{ij}.\]

\end{frame}

\begin{frame}
\frametitle{Getting rid of \(g\)}
Let \(\Lambda=\Lambda^i_{i'}=\left[\frac{\partial x^i}{\partial x^{i'}}\right]\), then:
\begin{equation}
\boldsymbol{g}'=\Lambda^T \boldsymbol{g} \Lambda
\end{equation}
Form (1) we get that 
\[g'=g\left(\det\Lambda\right)^2\]
and
\[
\boldsymbol{g}^{-1}=\Lambda(\boldsymbol{g}')^{-1}\Lambda^T
\]

so
\begin{align*}
\varepsilon^{ij}&=\pm\frac{1}{\sqrt{\gamma}}\sqrt{g}g^{ij}\\
		&=\pm\frac{1}{\sqrt{\gamma}}\frac{\sqrt{g'}}{|\det\Lambda|}\Lambda(\boldsymbol{g}')^{-1}\Lambda^T\\
		&=\frac{\sqrt{g'}}{\sqrt{\gamma}}\frac{\Lambda(\boldsymbol{g}')^{-1}\Lambda^T}{\det\Lambda}
\end{align*}
See \ref{theFormula}.
\end{frame}

\begin{frame}
\frametitle{Using the formula}
\[\begin{cases}
r= \frac{b-a}{b} r' + a\\
\theta = \theta'\\
z=z'
\end{cases}\]
\[
\frac{\sqrt{g'}}{\sqrt{\gamma}}\frac{\Lambda(\boldsymbol{g}')^{-1}\Lambda^T}{\det\Lambda}
\]
\[\Lambda=\text{diag}\left( \frac{b-a}{b} ,1,1\right)\]
\[\boldsymbol{g}'=\text{diag}(1,r'^2,1)\]
\[\boldsymbol{\gamma}=\text{diag}\left(1,r^2,1\right)\]
then
\[
\varepsilon^{ij}=\text{diag}
\left(
\frac{b-a}{b}\frac{r'}{r},
\frac{b}{rr'(b-a)},
\frac{r'b}{r(b-a)}
\right)
\]

\end{frame}

\begin{frame}

\begin{align*}
\varepsilon^i_j&=\varepsilon^{ik}\gamma_{kj}\\
&=\text{diag}

\right)
\]
\end{framed}
\end{frame}
