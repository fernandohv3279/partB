\begin{frame}
\frametitle{The main idea}
Consider the Helmholtz equation in $\mathbb{R}^2$
\begin{equation*}
	\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
and the change of coordinates \(u=3x,v=3y\).
\newline
\newline
\pause
What form does the Helmholtz equation take in the new coordinates \(u,v\)?
\pause
\begin{equation*}
\partial_x=\frac{du}{dx}\cdot\partial_u=3\partial_u\quad
\partial_y=\frac{dv}{dy}\cdot\partial_v=3\partial_v
\end{equation*}
\pause
Then the equation becomes:
\begin{equation*}
	\left(3^2\partial^2_u+3^2\partial^2_v+k^2\right)f=0
\end{equation*}
\begin{equation*}
	\highlightcyan{\left(\partial^2_u+\partial^2_v+(k/3)^2\right)f=0}
\end{equation*}
\begin{equation*}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{Less trivial example}
Consider the Helmholtz equation in $\mathbb{R}^2$
\begin{equation*}
	\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
and the change of coordinates \(u=\frac{x}{x^2+y^2}, v=\frac{y}{x^2+y^2}\).
\newline
\newline
\pause
What form does the Helmholtz equation take in the new coordinates \(u,v\)?
\pause
\begin{equation*}
	\partial_x=\dd{u}{x}\partial_u + \dd{v}{x}\partial_v \qquad
	\partial_y=\dd{u}{y}\partial_u + \dd{v}{y}\partial_v,
\end{equation*}
\pause
After some algebra the equation becomes:
\begin{equation*}
\left(
\left(\left(\dd{u}{x}\right)^2
+\left(\dd{u}{y}\right)^2\right)
\left(\partial_u^2+\partial_v^2\right)+ k^2\right)f=0
\end{equation*}
\begin{equation*}
\highlightcyan{\left(
	\partial_u^2+\partial_v^2
	+\left(
	k\middle/\sqrt{\left(\dd{u}{x}\right)^2
	+\left(\dd{u}{y}\right)^2}
	\right)^2
\right)f=0}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{Less trivial example}
Substituting
\begin{equation*}\label{partialu}
	\dd{u}{x}=\frac{y^2-x^2}{(x^2+y^2)^2} \qquad
	\dd{u}{y}=-\frac{2xy}{(x^2+y^2)^2},
\end{equation*}
in the equation
\begin{equation*}
\left(
	\partial_u^2+\partial_v^2
	+\left(
	k\middle/\sqrt{\left(\dd{u}{x}\right)^2
	+\left(\dd{u}{y}\right)^2}
	\right)^2
\right)f=0,
\end{equation*}
and doing some more algebra we get:
\begin{equation*}
\highlightcyan{\left(
	\partial_u^2
	+\partial_v^2
	+\left(k\left(x^2+y^2\right)\right)^2
\right)f=0}
\end{equation*}
\end{frame}

\begin{frame}
\frametitle{A non-example}
Not all transformations can be achieved by modifying \(k\).

Consider for example \( u=2x,v=3y\).
\pause

In this case:
\begin{equation*}
	\partial_x=2\partial_u \qquad
	\partial_y=3\partial_v,
\end{equation*}
\pause
and replacing in the original Helmholtz equation:
\begin{equation*}
\highlightcyan{\left(\partial^2_x+\partial^2_y+k^2\right)f=0,}
\end{equation*}
\pause

we get:
\begin{equation*}
	\highlightyellow{(4\partial_u^2 + 9\partial_v^2 + k^2)f=0,}
\end{equation*}
\pause
so we can not rewrite this equation to have the same form as the original
Helmholtz equation.
\end{frame}

\begin{frame}
\frametitle{In summary}
\begin{itemize}
\item We can encode some simple geometric transformations.
\item There are simple geometric transformations we can not encode.
\item We can encode some complicated (at least non-trivial) geometric transformations.
\item There are complicated geometric transformations we can not encode.
\end{itemize}
\pause

\centering
Coordinate transformation $\iff$ medium composition
\end{framed}
\end{frame}

\begin{frame}
\frametitle{Maxwell's equations}
\begin{align*}
\nabla \cdot E &= \frac{\rho}{\varepsilon_0}\\
\nabla \cdot B &= 0\\
\nabla \times E &= -\frac{\partial B}{\partial t}\\
\nabla \times B &= \frac{1}{c^2}\cdot\frac{\partial E}{\partial t} + \mu_0 J
\end{align*}
\pause
In a vacuum we have \(\varepsilon_0E=D\) and \(B=\mu_0H\), also
\(\mu_0\varepsilon_0=\frac{1}{c^2}\) which leads to:
\pause
\begin{align*}
\nabla \cdot D &= \rho\\
\nabla \cdot B &= 0\\
\nabla \times E &= -\frac{\partial B}{\partial t}\\
\nabla \times H &= \frac{\partial D}{\partial t} + J
\end{align*}
\end{frame}

\begin{frame}
\frametitle{Divergence}
\begin{equation*}
\nabla_{x,y}\cdot V = \partial_x V^x +\partial_y V^y \qquad\;\;\;\;\; \text{\emoji{slightly-smiling-face}}
\end{equation*}
\pause
\begin{equation*}
\nabla_{r,\theta}\cdot V = \partial_r V^r +\partial_\theta V^\theta \qquad\;\;\;\;\;\; \pause \text{\emoji{neutral-face}}
\end{equation*}
\pause
\begin{equation*}
	\nabla_{r,\theta}\cdot V = \frac{1}{r}\partial_r\left(rV^r\right) +\frac{1}{r}\partial_\theta V^\theta \quad \text{\emoji{slightly-smiling-face}}
\end{equation*}
\pause
\begin{equation*}
\nabla\cdot V = \frac{1}{\sqrt{g}}\partial_i\left(\sqrt{g}V^i\right) \qquad\quad\;\;\;\; \text{\emoji{star-struck}}
\end{equation*}

Here g is the determinant of the metric tensor.
\end{frame}
